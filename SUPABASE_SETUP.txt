安安打地鼠 v4｜Supabase 線上排行榜設定

1. 到 https://supabase.com 建立免費 Project。
2. 在 SQL Editor 執行：

create table public.anan_leaderboard (
  id bigint generated by default as identity primary key,
  name text not null check (char_length(name) between 1 and 16),
  score bigint not null default 0,
  created_at timestamptz not null default now()
);

alter table public.anan_leaderboard enable row level security;

create policy "public can read leaderboard"
on public.anan_leaderboard
for select
to anon
using (true);

create policy "public can insert leaderboard"
on public.anan_leaderboard
for insert
to anon
with check (char_length(name) between 1 and 16 and score >= 0);

3. Project Settings → API 取得 Project URL 與 anon public key。
4. 打開 index.html，搜尋 ONLINE_CONFIG，把兩個值填入：
const ONLINE_CONFIG={
  supabaseUrl:"https://xxxx.supabase.co",
  supabaseAnonKey:"你的 anon public key"
};

注意：不要使用 service_role key。
